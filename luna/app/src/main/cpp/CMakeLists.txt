cmake_minimum_required(VERSION 3.22)
project(luna LANGUAGES C CXX)
include(FetchContent)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
foreach(option LLAMA_BUILD_COMMON LLAMA_BUILD_EXAMPLES LLAMA_BUILD_TESTS LLAMA_BUILD_SERVER GGML_NATIVE GGML_OPENMP GGML_LLAMAFILE)
  set(${option} OFF CACHE BOOL "" FORCE)
endforeach()
FetchContent_Declare(engine GIT_REPOSITORY https://github.com/ggml-org/llama.cpp.git GIT_TAG b5046 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(engine)
add_library(luna SHARED engine.cpp)
target_compile_features(luna PRIVATE cxx_std_17)
target_link_libraries(luna PRIVATE llama android log)
